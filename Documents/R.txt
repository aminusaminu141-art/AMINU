CREATE TABLE INFOR (
id INT AUTO_INCREMENT PRIMARY KEY,
FISRT_NAME VARCHAR(20),
SURN_NAME VARCHAR(20),
LAST_NAME VARCHAR(20),
HOUSE_NUMBER INT(10),
STREE_NAME VARCHAR(20),
COUNTRY VARCHAR(30),
STATE VARCHAR(30),
LOCAL_GOVRMENT VARCHAR(30),
EMAIL VARCHAR(50),
PHONE_NUMBER INT(15),
GENDER VARCHAR(10)
);
